    USERS {
        Integer   id              PK
        String    email           UK "unique, indexed"
        String    password_hash
        String    role            "student | recruiter (host) | admin"
        Boolean   is_active
        DateTime  created_at
    }

    RECRUITER_PROFILES {
        Integer  id               PK
        Integer  user_id          FK,UK "→ users.id"
        String   full_name
        String   company_name
        String   company_logo
        String   designation
        String   phone
        String   company_website
        String   company_size
        String   industry
        Text     bio
        String   linkedin_url
        String   location
        DateTime updated_at
    }

    STUDENTS {
        Integer  id               PK
        Integer  user_id          FK,UK "→ users.id"
        String   full_name
        String   phone
        String   avatar_url
        String   gender           "male | female | other | prefer_not_to_say"
        String   university
        String   degree
        String   field_of_study
        Integer  graduation_year
        Float    cgpa
        Text     bio
        String   location
        String   preferred_mode   "Remote | Hybrid | In-Office"
        String   expected_stipend
        String   portfolio_url
        String   github_url
        String   linkedin_url
        DateTime updated_at
    }

    DOCUMENTS {
        Integer  id               PK
        Integer  student_id       FK,IDX "→ students.id"
        String   document_type    "resume | cover_letter | transcript | other"
        String   file_name
        String   file_url         "legacy external URL"
        String   content_type     "e.g. application/pdf"
        Integer  file_size        "bytes"
        MediumText file_data      "base64 payload (stored in DB)"
        DateTime uploaded_at
    }

    SKILLS {
        Integer  id               PK
        String   name             UK "unique, indexed"
        String   category         "Engineering | Design | Data Science | …"
    }

    STUDENT_SKILLS {
        Integer  id               PK
        Integer  student_id       FK,IDX "→ students.id"
        Integer  skill_id         FK,IDX "→ skills.id"
        String   proficiency_level "Beginner | Intermediate | Advanced"
    }

    INTERNSHIPS {
        Integer  id               PK
        Integer  recruiter_id     FK,IDX "→ recruiter_profiles.id (nullable)"
        String   title            IDX
        String   company_name     IDX
        String   company_logo
        String   location         IDX
        String   mode             "Remote | Hybrid | In-Office"
        Integer  stipend_amount
        String   stipend_currency "₹ | $ | €"
        String   duration
        Text     description
        Text     eligibility
        Text     responsibilities
        Text     benefits
        String   recruiter_name
        String   recruiter_email
        String   recruiter_title
        String   skills_required  "comma-separated"
        Boolean  is_external
        String   external_link
        Boolean  is_active
        DateTime deadline
        DateTime created_at
    }

    APPLICATIONS {
        Integer  id               PK
        Integer  student_id       FK,IDX "→ students.id"
        Integer  internship_id    FK,IDX "→ internships.id"
        String   status           "Applied | Under Review | Shortlisted | Interview Scheduled | Accepted | Rejected"
        Text     cover_letter
        MediumText resume_url     "URL or base64 data-URI upload"
        Text     recruiter_notes
        DateTime applied_at
    }

    INTERVIEWS {
        Integer  id               PK
        Integer  recruiter_id     FK "→ recruiter_profiles.id"
        Integer  application_id   FK "→ applications.id"
        Integer  internship_id    FK "→ internships.id (SET NULL)"
        DateTime scheduled_at
        Integer  duration_minutes
        String   interview_type   "Video Call | Phone Call | In-Person | Technical Test"
        String   meeting_link
        Text     notes
        String   status           "Scheduled | Completed | Cancelled"
        DateTime created_at
    }

    NOTIFICATIONS {
        Integer  id               PK
        Integer  user_id          FK,IDX "→ users.id"
        String   title
        Text     message
        String   type             "info | application | recommendation | alert"
        Boolean  is_read
        DateTime created_at
    }

    SAVED_INTERNSHIPS {
        Integer  id               PK
        Integer  student_id       FK,IDX "→ students.id"
        Integer  internship_id    FK,IDX "→ internships.id"
        DateTime saved_at
    }

    MESSAGES {
        Integer  id               PK
        Integer  sender_id        FK,IDX "→ users.id"
        Integer  receiver_id      FK,IDX "→ users.id"
        Text     content
        Boolean  is_read
        DateTime sent_at
    }
